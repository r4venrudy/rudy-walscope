cmake_minimum_required(VERSION 3.20)
project(rudy_walscope VERSION 0.1.0 LANGUAGES C)
option(WALSCOPE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(WALSCOPE_REQUIRE_PYTHON "Require the independent SQLite oracle suite" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/blake3/provenance.json" BLAKE3_PROVENANCE)
string(JSON BLAKE3_FILE_COUNT LENGTH "${BLAKE3_PROVENANCE}" files)
math(EXPR BLAKE3_FILE_LAST "${BLAKE3_FILE_COUNT} - 1")
foreach(index RANGE ${BLAKE3_FILE_LAST})
  string(JSON file_name GET "${BLAKE3_PROVENANCE}" files ${index} file)
  string(JSON expected_hash GET "${BLAKE3_PROVENANCE}" files ${index} sha256)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/blake3/${file_name}" actual_hash)
  if(NOT actual_hash STREQUAL expected_hash)
    message(FATAL_ERROR "Vendored BLAKE3 content differs from its pinned provenance: ${file_name}")
  endif()
endforeach()

add_library(blake3 STATIC third_party/blake3/blake3.c third_party/blake3/blake3_dispatch.c third_party/blake3/blake3_portable.c)
target_include_directories(blake3 SYSTEM PUBLIC third_party/blake3)
target_compile_definitions(blake3 PRIVATE BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512 BLAKE3_USE_NEON=0)
target_compile_features(blake3 PRIVATE c_std_11)
add_library(walscope_core src/io.c src/wal.c src/report.c)
target_include_directories(walscope_core PUBLIC include)
target_link_libraries(walscope_core PUBLIC blake3)
target_compile_features(walscope_core PUBLIC c_std_11)
add_executable(walscope src/main.c)
target_link_libraries(walscope PRIVATE walscope_core)
add_executable(walscope_tests tests/test_core.c)
target_link_libraries(walscope_tests PRIVATE walscope_core)

foreach(target walscope_core walscope walscope_tests)
  set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes -Werror)
    target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L _FILE_OFFSET_BITS=64)
    if(MINGW)
      target_link_options(${target} PRIVATE -static -static-libgcc)
    endif()
  endif()
endforeach()
if(MINGW)
  target_link_options(walscope PRIVATE -municode)
endif()
if(WALSCOPE_SANITIZE AND NOT MSVC)
  foreach(target blake3 walscope_core walscope walscope_tests)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()

enable_testing()
add_test(NAME native-core-and-hash-vectors COMMAND walscope_tests)
add_test(NAME cli-help COMMAND walscope --help)
add_test(NAME cli-version COMMAND walscope --version)
if(WALSCOPE_REQUIRE_PYTHON)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
else()
  find_package(Python3 3.9 COMPONENTS Interpreter)
endif()
if(Python3_FOUND)
  add_test(NAME sqlite-oracle-and-adversarial-captures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:walscope> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
install(TARGETS walscope RUNTIME DESTINATION bin)
install(FILES README.md LICENSE SECURITY.md CONTRIBUTING.md THIRD_PARTY.md DESTINATION .)
install(DIRECTORY docs examples assets DESTINATION . PATTERN "__pycache__" EXCLUDE)
install(FILES third_party/blake3/LICENSE_A2 third_party/blake3/LICENSE_A2LLVM third_party/blake3/LICENSE_CC0 DESTINATION licenses)
